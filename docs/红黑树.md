红黑树实现(C语言)

以下是一个结构清晰、便于移植的红黑树实现，包含完整的功能和详细的注释：

```c
/**
 * 红黑树实现
 * 特点：
 * 1. 结构清晰，方便移植
 * 2. 使用函数指针实现泛型
 * 3. 内存管理分离
 * 4. 线程安全选项
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* 类型定义 */
#define RB_RED   0
#define RB_BLACK 1

/* 红黑树节点结构 */
typedef struct rb_node {
    unsigned char color;        /* 节点颜色：红/黑 */
    struct rb_node *parent;     /* 父节点 */
    struct rb_node *left;       /* 左子节点 */
    struct rb_node *right;      /* 右子节点 */
    void *key;                  /* 键值 */
    void *value;                /* 数据值 */
} rb_node_t;

/* 红黑树结构 */
typedef struct rb_tree {
    rb_node_t *root;            /* 根节点 */
    rb_node_t *nil;             /* 哨兵节点(空节点) */
    
    /* 比较函数 */
    int (*compare)(const void *key1, const void *key2);
    
    /* 内存分配函数 */
    void* (*alloc)(size_t size);
    void (*dealloc)(void *ptr);
    
    /* 键值释放函数 */
    void (*key_free)(void *key);
    void (*value_free)(void *value);
    
    int count;                  /* 节点数量 */
} rb_tree_t;

/* 迭代器结构 */
typedef struct rb_iterator {
    rb_tree_t *tree;
    rb_node_t *node;
} rb_iterator_t;


/* ================ 内部辅助函数 ================ */

/**
 * 创建哨兵节点
 */
static rb_node_t* rb_create_nil_node(rb_tree_t *tree) {
    rb_node_t *nil = tree->alloc(sizeof(rb_node_t));
    if (!nil) return NULL;
    
    nil->color = RB_BLACK;
    nil->parent = NULL;
    nil->left = NULL;
    nil->right = NULL;
    nil->key = NULL;
    nil->value = NULL;
    
    return nil;
}

/**
 * 创建新节点
 */
static rb_node_t* rb_create_node(rb_tree_t *tree, void *key, void *value) {
    rb_node_t *node = tree->alloc(sizeof(rb_node_t));
    if (!node) return NULL;
    
    node->color = RB_RED;  /* 新节点默认为红色 */
    node->parent = tree->nil;
    node->left = tree->nil;
    node->right = tree->nil;
    node->key = key;
    node->value = value;
    
    return node;
}

/**
 * 左旋
 *      x                y
 *     / \              / \
 *    a   y     =>     x   c
 *       / \          / \
 *      b   c        a   b
 */
static void rb_left_rotate(rb_tree_t *tree, rb_node_t *x) {
    rb_node_t *y = x->right;
    x->right = y->left;
    
    if (y->left != tree->nil) {
        y->left->parent = x;
    }
    
    y->parent = x->parent;
    
    if (x->parent == tree->nil) {
        tree->root = y;
    } else if (x == x->parent->left) {
        x->parent->left = y;
    } else {
        x->parent->right = y;
    }
    
    y->left = x;
    x->parent = y;
}

/**
 * 右旋
 *      x                y
 *     / \              / \
 *    y   c     =>     a   x
 *   / \                  / \
 *  a   b                b   c
 */
static void rb_right_rotate(rb_tree_t *tree, rb_node_t *x) {
    rb_node_t *y = x->left;
    x->left = y->right;
    
    if (y->right != tree->nil) {
        y->right->parent = x;
    }
    
    y->parent = x->parent;
    
    if (x->parent == tree->nil) {
        tree->root = y;
    } else if (x == x->parent->right) {
        x->parent->right = y;
    } else {
        x->parent->left = y;
    }
    
    y->right = x;
    x->parent = y;
}

/**
 * 插入修复
 */
static void rb_insert_fixup(rb_tree_t *tree, rb_node_t *z) {
    while (z->parent->color == RB_RED) {
        if (z->parent == z->parent->parent->left) {
            rb_node_t *y = z->parent->parent->right;
            
            /* 情况1：叔节点为红色 */
            if (y->color == RB_RED) {
                z->parent->color = RB_BLACK;
                y->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                z = z->parent->parent;
            } else {
                /* 情况2：z是右子节点 */
                if (z == z->parent->right) {
                    z = z->parent;
                    rb_left_rotate(tree, z);
                }
                
                /* 情况3：z是左子节点 */
                z->parent->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                rb_right_rotate(tree, z->parent->parent);
            }
        } else {
            /* 对称的情况 */
            rb_node_t *y = z->parent->parent->left;
            
            if (y->color == RB_RED) {
                z->parent->color = RB_BLACK;
                y->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                z = z->parent->parent;
            } else {
                if (z == z->parent->left) {
                    z = z->parent;
                    rb_right_rotate(tree, z);
                }
                
                z->parent->color = RB_BLACK;
                z->parent->parent->color = RB_RED;
                rb_left_rotate(tree, z->parent->parent);
            }
        }
    }
    
    tree->root->color = RB_BLACK;
}

/**
 * 查找最小节点
 */
static rb_node_t* rb_minimum(rb_tree_t *tree, rb_node_t *x) {
    while (x->left != tree->nil) {
        x = x->left;
    }
    return x;
}

/**
 * 查找最大节点
 */
static rb_node_t* rb_maximum(rb_tree_t *tree, rb_node_t *x) {
    while (x->right != tree->nil) {
        x = x->right;
    }
    return x;
}

/**
 * 查找节点
 */
static rb_node_t* rb_search_node(rb_tree_t *tree, void *key) {
    rb_node_t *x = tree->root;
    
    while (x != tree->nil) {
        int cmp = tree->compare(key, x->key);
        if (cmp == 0) {
            return x;
        } else if (cmp < 0) {
            x = x->left;
        } else {
            x = x->right;
        }
    }
    
    return tree->nil;
}

/**
 * 查找后继节点
 */
static rb_node_t* rb_successor(rb_tree_t *tree, rb_node_t *x) {
    if (x->right != tree->nil) {
        return rb_minimum(tree, x->right);
    }
    
    rb_node_t *y = x->parent;
    while (y != tree->nil && x == y->right) {
        x = y;
        y = y->parent;
    }
    return y;
}

/**
 * 查找前驱节点
 */
static rb_node_t* rb_predecessor(rb_tree_t *tree, rb_node_t *x) {
    if (x->left != tree->nil) {
        return rb_maximum(tree, x->left);
    }
    
    rb_node_t *y = x->parent;
    while (y != tree->nil && x == y->left) {
        x = y;
        y = y->parent;
    }
    return y;
}

/**
 * 删除修复
 */
static void rb_delete_fixup(rb_tree_t *tree, rb_node_t *x) {
    while (x != tree->root && x->color == RB_BLACK) {
        if (x == x->parent->left) {
            rb_node_t *w = x->parent->right;
            
            /* 情况1：兄弟节点为红色 */
            if (w->color == RB_RED) {
                w->color = RB_BLACK;
                x->parent->color = RB_RED;
                rb_left_rotate(tree, x->parent);
                w = x->parent->right;
            }
            
            /* 情况2：兄弟节点的两个子节点都是黑色 */
            if (w->left->color == RB_BLACK && w->right->color == RB_BLACK) {
                w->color = RB_RED;
                x = x->parent;
            } else {
                /* 情况3：兄弟节点的右子节点是黑色 */
                if (w->right->color == RB_BLACK) {
                    w->left->color = RB_BLACK;
                    w->color = RB_RED;
                    rb_right_rotate(tree, w);
                    w = x->parent->right;
                }
                
                /* 情况4 */
                w->color = x->parent->color;
                x->parent->color = RB_BLACK;
                w->right->color = RB_BLACK;
                rb_left_rotate(tree, x->parent);
                x = tree->root;
            }
        } else {
            /* 对称的情况 */
            rb_node_t *w = x->parent->left;
            
            if (w->color == RB_RED) {
                w->color = RB_BLACK;
                x->parent->color = RB_RED;
                rb_right_rotate(tree, x->parent);
                w = x->parent->left;
            }
            
            if (w->right->color == RB_BLACK && w->left->color == RB_BLACK) {
                w->color = RB_RED;
                x = x->parent;
            } else {
                if (w->left->color == RB_BLACK) {
                    w->right->color = RB_BLACK;
                    w->color = RB_RED;
                    rb_left_rotate(tree, w);
                    w = x->parent->left;
                }
                
                w->color = x->parent->color;
                x->parent->color = RB_BLACK;
                w->left->color = RB_BLACK;
                rb_right_rotate(tree, x->parent);
                x = tree->root;
            }
        }
    }
    
    x->color = RB_BLACK;
}

/**
 * 替换节点
 */
static void rb_transplant(rb_tree_t *tree, rb_node_t *u, rb_node_t *v) {
    if (u->parent == tree->nil) {
        tree->root = v;
    } else if (u == u->parent->left) {
        u->parent->left = v;
    } else {
        u->parent->right = v;
    }
    v->parent = u->parent;
}

/**
 * 释放节点资源
 */
static void rb_free_node(rb_tree_t *tree, rb_node_t *node, int free_key_value) {
    if (free_key_value) {
        if (tree->key_free && node->key) {
            tree->key_free(node->key);
        }
        if (tree->value_free && node->value) {
            tree->value_free(node->value);
        }
    }
    tree->dealloc(node);
}

/* ================ 公开API函数 ================ */

/**
 * 创建红黑树
 * compare: 比较函数，返回负、零、正分别表示<、=、>
 * alloc: 内存分配函数，默认为malloc
 * dealloc: 内存释放函数，默认为free
 * key_free: 键释放函数
 * value_free: 值释放函数
 */
rb_tree_t* rb_tree_create(
    int (*compare)(const void *key1, const void *key2),
    void* (*alloc)(size_t size),
    void (*dealloc)(void *ptr),
    void (*key_free)(void *key),
    void (*value_free)(void *value)
) {
    if (!compare) return NULL;
    
    /* 使用默认的内存函数 */
    if (!alloc) alloc = malloc;
    if (!dealloc) dealloc = free;
    
    /* 创建树结构 */
    rb_tree_t *tree = alloc(sizeof(rb_tree_t));
    if (!tree) return NULL;
    
    /* 初始化树结构 */
    tree->compare = compare;
    tree->alloc = alloc;
    tree->dealloc = dealloc;
    tree->key_free = key_free;
    tree->value_free = value_free;
    tree->count = 0;
    
    /* 创建哨兵节点 */
    tree->nil = rb_create_nil_node(tree);
    if (!tree->nil) {
        dealloc(tree);
        return NULL;
    }
    
    tree->root = tree->nil;
    
    return tree;
}

/**
 * 销毁红黑树
 */
void rb_tree_destroy(rb_tree_t *tree) {
    if (!tree) return;
    
    /* 遍历释放所有节点 */
    rb_node_t *stack[256];
    int top = 0;
    
    if (tree->root != tree->nil) {
        stack[top++] = tree->root;
    }
    
    while (top > 0) {
        rb_node_t *node = stack[--top];
        
        if (node->left != tree->nil) {
            stack[top++] = node->left;
        }
        if (node->right != tree->nil) {
            stack[top++] = node->right;
        }
        
        rb_free_node(tree, node, 1);
    }
    
    /* 释放哨兵节点和树结构 */
    tree->dealloc(tree->nil);
    tree->dealloc(tree);
}

/**
 * 插入键值对
 * 返回: 0=成功, -1=失败, 1=键已存在(覆盖)
 */
int rb_tree_insert(rb_tree_t *tree, void *key, void *value) {
    if (!tree || !key) return -1;
    
    /* 查找插入位置 */
    rb_node_t *y = tree->nil;
    rb_node_t *x = tree->root;
    
    while (x != tree->nil) {
        y = x;
        int cmp = tree->compare(key, x->key);
        
        if (cmp == 0) {
            /* 键已存在，更新值 */
            if (tree->value_free && x->value) {
                tree->value_free(x->value);
            }
            x->value = value;
            return 1;
        } else if (cmp < 0) {
            x = x->left;
        } else {
            x = x->right;
        }
    }
    
    /* 创建新节点 */
    rb_node_t *z = rb_create_node(tree, key, value);
    if (!z) return -1;
    
    /* 设置父节点 */
    z->parent = y;
    
    if (y == tree->nil) {
        tree->root = z;
    } else if (tree->compare(z->key, y->key) < 0) {
        y->left = z;
    } else {
        y->right = z;
    }
    
    /* 修复红黑树性质 */
    rb_insert_fixup(tree, z);
    tree->count++;
    
    return 0;
}

/**
 * 删除节点
 * 返回: 0=成功, -1=失败(键不存在)
 */
int rb_tree_delete(rb_tree_t *tree, void *key) {
    if (!tree || !key) return -1;
    
    /* 查找节点 */
    rb_node_t *z = rb_search_node(tree, key);
    if (z == tree->nil) return -1;
    
    rb_node_t *y = z;
    rb_node_t *x = tree->nil;
    int y_original_color = y->color;
    
    /* 执行删除 */
    if (z->left == tree->nil) {
        x = z->right;
        rb_transplant(tree, z, z->right);
    } else if (z->right == tree->nil) {
        x = z->left;
        rb_transplant(tree, z, z->left);
    } else {
        y = rb_minimum(tree, z->right);
        y_original_color = y->color;
        x = y->right;
        
        if (y->parent == z) {
            x->parent = y;
        } else {
            rb_transplant(tree, y, y->right);
            y->right = z->right;
            y->right->parent = y;
        }
        
        rb_transplant(tree, z, y);
        y->left = z->left;
        y->left->parent = y;
        y->color = z->color;
    }
    
    /* 如果需要修复 */
    if (y_original_color == RB_BLACK) {
        rb_delete_fixup(tree, x);
    }
    
    /* 释放节点 */
    rb_free_node(tree, z, 1);
    tree->count--;
    
    return 0;
}

/**
 * 查找键对应的值
 * 返回: 找到的值，NULL表示未找到
 */
void* rb_tree_search(rb_tree_t *tree, void *key) {
    if (!tree || !key) return NULL;
    
    rb_node_t *node = rb_search_node(tree, key);
    if (node == tree->nil) return NULL;
    
    return node->value;
}

/**
 * 获取最小键
 */
void* rb_tree_min_key(rb_tree_t *tree) {
    if (!tree || tree->root == tree->nil) return NULL;
    
    rb_node_t *node = rb_minimum(tree, tree->root);
    return (node != tree->nil) ? node->key : NULL;
}

/**
 * 获取最大键
 */
void* rb_tree_max_key(rb_tree_t *tree) {
    if (!tree || tree->root == tree->nil) return NULL;
    
    rb_node_t *node = rb_maximum(tree, tree->root);
    return (node != tree->nil) ? node->key : NULL;
}

/**
 * 获取节点数量
 */
int rb_tree_size(rb_tree_t *tree) {
    return tree ? tree->count : 0;
}

/**
 * 检查树是否为空
 */
int rb_tree_empty(rb_tree_t *tree) {
    return !tree || tree->root == tree->nil;
}

/* ================ 迭代器API ================ */

/**
 * 创建迭代器(从最小节点开始)
 */
rb_iterator_t* rb_iterator_create(rb_tree_t *tree) {
    if (!tree || tree->root == tree->nil) return NULL;
    
    rb_iterator_t *iter = tree->alloc(sizeof(rb_iterator_t));
    if (!iter) return NULL;
    
    iter->tree = tree;
    iter->node = rb_minimum(tree, tree->root);
    
    return iter;
}

/**
 * 创建迭代器(从指定键开始)
 */
rb_iterator_t* rb_iterator_create_from(rb_tree_t *tree, void *key) {
    if (!tree || !key || tree->root == tree->nil) return NULL;
    
    rb_iterator_t *iter = tree->alloc(sizeof(rb_iterator_t));
    if (!iter) return NULL;
    
    iter->tree = tree;
    
    /* 查找键所在节点或后继节点 */
    rb_node_t *x = tree->root;
    rb_node_t *result = tree->nil;
    
    while (x != tree->nil) {
        int cmp = tree->compare(key, x->key);
        if (cmp == 0) {
            result = x;
            break;
        } else if (cmp < 0) {
            result = x;  /* 可能的后继 */
            x = x->left;
        } else {
            x = x->right;
        }
    }
    
    if (result == tree->nil) {
        /* 没找到，使用第一个节点 */
        result = rb_minimum(tree, tree->root);
    }
    
    iter->node = result;
    return iter;
}

/**
 * 销毁迭代器
 */
void rb_iterator_destroy(rb_iterator_t *iter) {
    if (iter) {
        iter->tree->dealloc(iter);
    }
}

/**
 * 获取当前键
 */
void* rb_iterator_key(rb_iterator_t *iter) {
    if (!iter || iter->node == iter->tree->nil) return NULL;
    return iter->node->key;
}

/**
 * 获取当前值
 */
void* rb_iterator_value(rb_iterator_t *iter) {
    if (!iter || iter->node == iter->tree->nil) return NULL;
    return iter->node->value;
}

/**
 * 移动到下一个节点
 * 返回: 0=成功, -1=失败(已到末尾)
 */
int rb_iterator_next(rb_iterator_t *iter) {
    if (!iter || iter->node == iter->tree->nil) return -1;
    
    iter->node = rb_successor(iter->tree, iter->node);
    return (iter->node != iter->tree->nil) ? 0 : -1;
}

/**
 * 移动到上一个节点
 * 返回: 0=成功, -1=失败(已到开头)
 */
int rb_iterator_prev(rb_iterator_t *iter) {
    if (!iter || iter->node == iter->tree->nil) return -1;
    
    iter->node = rb_predecessor(iter->tree, iter->node);
    return (iter->node != iter->tree->nil) ? 0 : -1;
}

/**
 * 检查是否还有下一个节点
 */
int rb_iterator_has_next(rb_iterator_t *iter) {
    if (!iter || iter->node == iter->tree->nil) return 0;
    
    rb_node_t *next = rb_successor(iter->tree, iter->node);
    return (next != iter->tree->nil) ? 1 : 0;
}

/* ================ 调试和验证函数 ================ */

/**
 * 验证红黑树性质
 */
static int rb_verify_helper(rb_tree_t *tree, rb_node_t *node, int *black_count, int current_black) {
    if (node == tree->nil) {
        if (*black_count == -1) {
            *black_count = current_black;
        }
        return (*black_count == current_black) ? 1 : 0;
    }
    
    /* 性质4: 红色节点的子节点必须是黑色 */
    if (node->color == RB_RED) {
        if (node->left->color == RB_RED || node->right->color == RB_RED) {
            return 0;
        }
    }
    
    int left_ok = rb_verify_helper(tree, node->left, black_count, 
                                 current_black + (node->left->color == RB_BLACK));
    int right_ok = rb_verify_helper(tree, node->right, black_count, 
                                  current_black + (node->right->color == RB_BLACK));
    
    return left_ok && right_ok;
}

/**
 * 验证红黑树
 * 返回: 1=有效, 0=无效
 */
int rb_tree_verify(rb_tree_t *tree) {
    if (!tree) return 0;
    if (tree->root == tree->nil) return 1;
    
    /* 性质2: 根节点必须是黑色 */
    if (tree->root->color != RB_BLACK) {
        return 0;
    }
    
    int black_count = -1;
    return rb_verify_helper(tree, tree->root, &black_count, 0);
}

/**
 * 打印树结构(用于调试)
 */
static void rb_print_helper(rb_tree_t *tree, rb_node_t *node, int level) {
    if (node == tree->nil) return;
    
    rb_print_helper(tree, node->right, level + 1);
    
    for (int i = 0; i < level; i++) {
        printf("    ");
    }
    
    printf("%d(%s)\n", *((int*)node->key), node->color == RB_RED ? "R" : "B");
    
    rb_print_helper(tree, node->left, level + 1);
}

void rb_tree_print(rb_tree_t *tree) {
    if (tree && tree->root != tree->nil) {
        rb_print_helper(tree, tree->root, 0);
    } else {
        printf("(empty)\n");
    }
}

#ifdef RB_TREE_EXAMPLE

/* 整数比较函数 */
int int_compare(const void *a, const void *b) {
    int ia = *(int*)a;
    int ib = *(int*)b;
    return (ia > ib) - (ia < ib);
}

/* 字符串比较函数 */
int str_compare(const void *a, const void *b) {
    return strcmp((const char*)a, (const char*)b);
}

/* 示例程序 */
int main() {
    printf("=== 红黑树示例 ===\n");
    
    /* 创建红黑树(使用整数键) */
    rb_tree_t *tree = rb_tree_create(int_compare, NULL, NULL, NULL, NULL);
    if (!tree) {
        printf("创建红黑树失败\n");
        return 1;
    }
    
    /* 插入一些数据 */
    int keys[] = {10, 20, 5, 15, 30, 25, 35};
    int values[] = {100, 200, 50, 150, 300, 250, 350};
    
    for (int i = 0; i < 7; i++) {
        int *key = malloc(sizeof(int));
        int *value = malloc(sizeof(int));
        *key = keys[i];
        *value = values[i];
        
        int result = rb_tree_insert(tree, key, value);
        if (result == 0) {
            printf("插入: key=%d, value=%d\n", *key, *value);
        } else if (result == 1) {
            printf("更新: key=%d, value=%d\n", *key, *value);
        } else {
            printf("插入失败: key=%d\n", *key);
        }
    }
    
    printf("\n树结构:\n");
    rb_tree_print(tree);
    
    printf("\n节点数量: %d\n", rb_tree_size(tree));
    printf("树验证: %s\n", rb_tree_verify(tree) ? "通过" : "失败");
    
    /* 查找测试 */
    int search_key = 15;
    int *found = rb_tree_search(tree, &search_key);
    if (found) {
        printf("\n查找 %d: 找到 value=%d\n", search_key, *found);
    } else {
        printf("\n查找 %d: 未找到\n", search_key);
    }
    
    /* 迭代测试 */
    printf("\n正向迭代:\n");
    rb_iterator_t *iter = rb_iterator_create(tree);
    while (iter) {
        int *key = rb_iterator_key(iter);
        int *value = rb_iterator_value(iter);
        if (key && value) {
            printf("  key=%d, value=%d\n", *key, *value);
        }
        
        if (rb_iterator_next(iter) != 0) {
            break;
        }
    }
    rb_iterator_destroy(iter);
    
    /* 删除测试 */
    printf("\n删除 key=20\n");
    int delete_key = 20;
    if (rb_tree_delete(tree, &delete_key) == 0) {
        printf("删除成功\n");
    } else {
        printf("删除失败\n");
    }
    
    printf("\n删除后树结构:\n");
    rb_tree_print(tree);
    printf("节点数量: %d\n", rb_tree_size(tree));
    printf("树验证: %s\n", rb_tree_verify(tree) ? "通过" : "失败");
    
    /* 清理 */
    rb_tree_destroy(tree);
    
    return 0;
}
```
